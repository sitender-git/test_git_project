\section{Application to three point computation}
The amplitude will be given by the terms that are at cubic order in the singularity. Thus, we need to pick the single pole term in the OPE of $U_1\tilde{U}_{23}$ and the quadratic piece in the OPE of $U_1\hat{U}_{23}$. So, far we have not made use of the sYM e.o.m. But, now we make use of these and the kinematic constraints for massless states in three point amplitudes i.e.  $k_i\cd k_j=0$. 
For the first case we have 
\be 
U_1(z_1) \tilde{U}^{23}(z_3)&=&:\Pi^m A^1_m: \tilde{U}^{23}(z_3) + :d_\a W^\a_1: \tilde{U}^{23}(z_3) \non\\
&=&  \f{-k^m_{23} A^{1}_{m} \tilde{U}^{23}}{z_{13}} -\f{W_1^{\a} D_\a\tilde{U}^{23}}{z_{13}}+ \text{regular} \non\\&=&\f{k^m_{1} A^{1}_{m} \tilde{U}^{23}}{z_{13}}-\f{W_1^{\a} D_\a\tilde{U}^{23}}{z_{13}}+ \text{regular} 
\ee
Hence the relevant piece for the amplitude is 
\be 
S^I_{1(23)}\equiv-\f{W_1^{\a} D_\a\tilde{U}^{23}}{z_{13} z^2_{23}}
\ee


Further using the result in \eqref{} we find that by noting ($i\rightarrow 1, j\rightarrow 23$)

\be 
\f{U_1(z_1) \hat{U}^{23}(z_3)}{z_{23}}&=&\f{1}{z_{23}z^2_{13}}\biggl[-\blue{\rb{k_{23}\cdot A_{1}}}\rb{k_1\cdot A_{23}}-\rb{A_1 \cdot A_{23}} + D_\a\blue{\rb{k_{23}\cdot A^{1}}} W_{23}^{\a} -W^\a_{1} D_\a\rb{k_{1}\cd A_{23}} \non\\
	&&+D_\b W_{1}^\a D_\a W_{23}^\b +\blue{k^{23}_m\rb{W_{1}\g^m W_{23}}}-\rb{W_{1} A_{23}}+\rb{A^{1} W_{23}} +\f{3}{2}F^{1}_{mn}F^{mn}_{23}\biggl] \non\\
	&=&\f{1}{z_{23}z^2_{13}}\biggl[-\rb{A_1 \cdot A_{23}}  -W^\a_{1} D_\a\rb{k_{1}\cd A_{23}} +D_\b W_{1}^\a D_\a W_{23}^\b -\rb{W_{1} A_{23}} \non\\
	&&+\rb{A^{1} W_{23}} +\f{3}{2}F^{1}_{mn}F^{mn}_{23}\biggl] \label{contri_2}
\ee
This, contribution we shall call $S^{II}_{1(23)}$ and thus, we need to calculate

\be 
S^{II}_{1(23)}&\equiv& \f{1}{z_{23}z^2_{13}}\biggl[-\rb{A_1 \cdot A_{23}}  -W^\a_{1} D_\a\rb{k_{1}\cd A_{23}} +D_\b W_{1}^\a D_\a W_{23}^\b -\rb{W_{1} A_{23}} \non\\
&&+\rb{A^{1} W_{23}} +\f{3}{2}F^{1}_{mn}F^{mn}_{23}\biggl] 
\ee

Let us calculate each of the contributions one by one.

\subsection{$S^I_{1(23)}$} 

We note that
\be 
D_\a \tilde{U}^{ij}&=&D_\a\rb{ -\rb{k_j\cdot A_i}\rb{k_i\cdot A_j}-\rb{A_i\cdot A_j} + D_\b\rb{k_j\cdot A^i} W_j^{\b} -W^\b_i D_\b\rb{k_i\cd A_j}} \non\\
&&+D_\a\rb{D_\b W_i^\s D_\s W_j^\b +k^{ij}_m\rb{W_i\g^m W_j}-\rb{W_i A_j}+\rb{A^i W_j} +\f{3}{2}F^i_{mn}F^{mn}_j}\non\\
&=&-\rb{k_j\cdot D_\a A_i}\rb{k_i\cdot A_j} -\rb{k_j\cdot  A_i}\rb{k_i\cdot D_\a A_j}-\rb{D_\a A_i\cdot A_j} -\rb{ A_i\cdot D_\a  A_j} \non\\
&&+ D_\a D_\b\rb{k_j\cdot A^i} W_j^{\b} -D_\b\rb{k_j\cdot A^i} D_\a W_j^{\b}- D_\a W^\b_i D_\b\rb{k_i\cd A_j} + W^\b_i D_\b\rb{k_i\cd D_\a A_j}\non\\
&&+D_\a D_\b W_i^\s D_\s W_j^\b + D_\b W_i^\s  D_\a D_\s W_j^\b-\rb{D_\a W_i A_j} +\rb{W_i D_\a A_j}\non\\
&&+\rb{D_\a A^i W_j} -\rb{A^i D_\a W_j}+\f{3}{2}D_\a F^i_{mn}F^{mn}_j +\f{3}{2} F^i_{mn} D_\a F^{mn}_j\non\\
&=&-k^j_m\rb{\g^m W_i}_\a\rb{k_i\cdot A_j} -\rb{k_j\cdot  A_i}k^i_m\rb{\g^m W_j}_\a-\rb{\g^m W_i}_\a A^j_m -A^i_\a\rb{k_i\cd A_j}\non\\
&&-A^i_m\rb{\g^m W_j}_\a  -\rb{k_j\cd A_i}A^j_\a +\f{1}{4} k^j_m F^i_{st} \rb{\g^{st} \g^m W_j}_{\a} - \f{1}{4} k^j_m \rb{\g^{st} \g^m W_i}_{\a} F^j_{st} \non\\
&&- \f{1}{4}F^i_{st}k^i_m\rb{\g^{st}\g^m W_j}_\a -\f{1}{4}k^i_m \rb{\g^m \g^{st} W_i}_\a F^j_{st} +4k_j^t (\g_t W^i)_\a \rb{k_i \cd A_j} \non\\
&&-\f{1}{4}F^i_{mn}\rb{\g^{mn} A_j}_\a +\rb{W_i D_\a A_j} + \rb{D_\a A^i W_j} -\f{1}{4}\rb{\g_{mn} A_i}_\a F^i_{mn}\non\\
&&+3 k^i_m (\g_n W_i)_\a F^{mn}_j + 3 F^i_{mn} k_j^{m}\rb{\g^n W_j}_\a 
\ee


Thus, we find that 
\be 
W^{\a} D_\a\tilde{U}^{23}&=& -k^3_m\rb{W_1\g^m W_2}\rb{k_2\cdot A_3} -\rb{k_3\cdot  A_2}k^2_m\rb{W_1\g^m W_3}-\rb{W_1\g^m W_2} A^3_m -\rb{W_1A^2} \rb{k_2\cd A_3}\non\\
&&-A^2_m\rb{W_1\g^m W_3}  -\rb{k_3\cd A_2}\rb{W^1A^3} +\f{1}{4} k^3_m F^2_{st} \rb{W_1\g^{st} \g^m W_3} - \f{1}{4} k^3_m \rb{W_1\g^{st} \g^m W_2} F^3_{st} \non\\
&&- \f{1}{4}F^2_{st}k^2_m\rb{W_1\g^{st}\g^m W_3} -\f{1}{4}k^2_m \rb{W_1\g^m \g^{st} W_2} F^3_{st} +4k_3^t (W_1\g_t W^2)\rb{k_2 \cd A_3} \non\\
&&-\f{1}{4}F^2_{mn}\rb{W_1\g^{mn} A_3} +W^\a_1\rb{W_2 D_\a A_3} + W_1^\a \rb{D_\a A^2 W_3} +\f{1}{4}\rb{W_1\g_{mn} A_2} F^3_{mn}\non\\
&&+3 k^2_m (W_1\g_n W_2) F^{mn}_3 + 3 F^2_{mn} k_3^{m}\rb{W_1\g^n W_3}\non\\
&=&-\rb{W_1\g^m W_2} A^3_m -\rb{W_1A^2} \rb{k_2\cd A_3}-A^2_m\rb{W_1\g^m W_3}  -\rb{k_3\cd A_2}\rb{W^1A^3}   \non\\
&&-\f{1}{4}F^2_{mn}\rb{W_1\g^{mn} A_3} +W^\a_1\rb{W_2 D_\a A_3} + W_1^\a \rb{D_\a A^2 W_3} -\f{1}{4}\rb{W_1\g_{mn} A_2} F^3_{mn}
\ee
where, the last line vanishes due to
\be 
k^2_m (W_1\g_n W_2) F^{mn}_3= \rb{k_2\cd k_3} (W_1\g_n W_2) A_3^n -  k^n_3(W_1\g_n W_2)=0 
\ee

Thus, we have 

\be 
-W_1^{\a} D_\a\tilde{U}^{23} &=&\rb{W_1\g^m W_2} A^3_m +\rb{W_1A^2} \rb{k_2\cd A_3}+A^2_m\rb{W_1\g^m W_3}  +\rb{k_3\cd A_2}\rb{W^1A^3}   \non\\
&&+\f{1}{4}F^2_{mn}\rb{W_1\g^{mn} A_3} -W^\a_1\rb{W_2 D_\a A_3} - W_1^\a \rb{D_\a A^2 W_3} +\f{1}{4}\rb{W_1\g_{mn} A_2} F^3_{mn} \non
\ee
Thus, we find that 
\be 
S^I_{1(23)}&=&\f{1}{z_{13} z^2_{23}} \biggl[\rb{W_1\g^m W_2} A^3_m +\rb{W_1A^2} \rb{k_2\cd A_3}+A^2_m\rb{W_1\g^m W_3}  +\rb{k_3\cd A_2}\rb{W^1A^3}   \non\\
&&+\f{1}{4}F^2_{mn}\rb{W_1\g^{mn} A_3} -W^\a_1\rb{W_2 D_\a A_3} - W_1^\a \rb{D_\a A^2 W_3} +\f{1}{4}\rb{W_1\g_{mn} A_2} F^3_{mn} \biggl]
\ee
